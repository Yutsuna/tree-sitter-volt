# Regression: FunctionalLowering must not lose a rewrite depending on where
# the expression arena happens to reallocate.
#
# Sections nested inside call arguments used to be rewritten through a field
# reference borrowed from the live parent node; every Context.Add() inside
# LowerSection can reallocate the arena, so the write landed in freed memory
# and the Section survived the Lowering phase — or not, depending on the file
# size. See ArenaStabilityPadded.vl for the same program shifted by
# four unrelated defs: both must lower identically.
#
# Expected: zero Section and zero Composition nodes in `parse --lowered`.

  def self.run -> Void
    users = [ " alice ", "   " ]

    kept = users.filter( (&.empty?.!) )
    tidy = users.map( (&.trim) >> (&.prefix("usr_")) )
  end

end

  end

samples/Sema/Assignability.vl
==================
# The five value positions that must agree with the type they flow into.
# Before phase C not one of them compared anything: `def r -> Int32 ; "plain" ;
# end` type-checked cleanly. All five now go through the single predicate
# `TypeCompat::IsAssignable`, called after constraint propagation so that an
# unconstrained literal has already been narrowed to the expected type.
#
# Everything here is *valid* — the five refusals live in the
# AssignMismatch*.vl samples.

# 1. Local declaration with a written type.
def local_decl -> Int32
  n : Int32 = 42
  wide : UInt64 = 7
  text : String = "ok"
  n
end

# 2. Assignment to a target whose type is known — local, @ivar, or member.
struct Counter
  getter total : UInt64

  def initialize
    @total = 0
  end

  def bump( by : UInt64 ) -> Void
    @total = @total + by
    local : UInt64 = 1
    local = local + by
  end
end

# 3. `return` against the declared return type.
def explicit_return -> String
  return "done"
end

# 4. A method's trailing expression, same rule as `return`.
def trailing -> Bool
  1 == 1
end

# A `-> Void` method's return type is an invalid SemaTypeId, so its trailing
# value is checked against nothing — no name is compared to reach that.
def void_discards_its_tail -> Void
  99
end

# 5. Parameter defaults.
def defaulted( n : Int32 = 5, label : String = "x", wide : UInt64 = 1 ) -> Int32
  n
end

# Widening between scalars of one family is allowed, and is what lets every
# integer width answer `Hashable`'s `hash -> UInt64` contract by returning
# `self`. Narrowing is not — see AssignMismatchLocal.vl.
def widening -> UInt64
  small : Int8 = 3
  small
end

# `null` fits any pointer-shaped slot; it is recognised through the node kind
# `Nil` claims, never through its name.
def nil_into_pointer -> Void
  buf : Pointer<UInt8> = Pointer<UInt8>.malloc( 8 )
  if buf != nil
    buf.free
  end
end

# A `do | x |` parameter carries no annotation: it takes its type from the
# `&block` slot the callee declares. An unqualified `each` inside a method
# body is a call on `self`, so it resolves — and types its block — exactly
# like `self.each` would.

struct Counter
  getter limit : Int32

  def initialize( limit : Int32 = 3 )
    @limit = limit
  end

  def each( &block : Int32 -> Void ) -> Void
    i = 0
    while i < @limit
      block.call( i )
      i += 1
    end
  end

  def sum -> Int32
    total = 0
    each do | value |
      total += value
    end
    total
  end
end

def block_param_types -> Void
  c = Counter.new( 5 )
  c.each do | value |
    doubled : Int32 = value * 2
  end
  total : Int32 = c.sum

  items = [ 1, 2, 3 ]
  items.each do | item |
    squared : Int32 = item * item
  end

  # `each_with_index` declares `&block : ( T, UInt64 ) -> Void` — two
  # positional slots on the same `&block`, locking in that a block's
  # arity is not limited to one parameter.
  items.each_with_index do | item, index |
    product : UInt64 = index * 2_u64
    doubled_item : Int32 = item * 2
  end
end

# A mixin included with an argument keeps that argument: `Enumerable<T>` on an
# `Array<T>` is what lets `map` see the element type, and a method generic
# declared on `map<U>` is what lets its result follow the block.

mixin Boxed<T>
  abstract def get -> T

  def get_twice -> T
    get
  end

  def convert<U>( &block : T -> U ) -> Array<U>
    result = Array<U>.new
    result.push( block.call( get ) )
    result
  end
end

  end

# A bare `IntLiteral` is typed late, from whatever context consumes it
# (Point 0 in PLAN.md). This sample extends the same late/lazy inference to
# `FloatLiteral` and to naked collection literals (`ArrayLit`, `HashLit`):
# neither has its own opinion about its element/key/value types, so the
# target type of a typed local, a return, or a call argument must propagate
# down through `ConstrainExprType` — same mechanism, same code path.

def take_float( value : Float32 ) -> Float32
  value
end

# A naked float literal defaults to Float64; the parameter type (Float32)
# must retroactively narrow it, exactly like an Int32 default narrows to
# UInt64 for an integer literal.
def float_literal_inference -> Float32
  take_float( 2.5 )
end

def sum_all( values : Array<UInt64> ) -> UInt64
  total : UInt64 = 0
  for value in values
    total += value
  end
  total
end

def array_literal_inference -> UInt64

  # The array literal itself carries no element type — each `IntLiteral`
  # inside it must inherit `UInt64` from the declared local type.
  fixed : Array<UInt64> = [ 1, 2, 3 ]

  # Passed directly as a call argument: the parameter type constrains the
  # literal array (and, through it, every element) at the call site.
  via_call : UInt64 = sum_all( [ 4, 5, 6 ] )

  fixed.size + via_call
end

def hash_literal_inference -> UInt64

  # A hash literal propagates its declared value type into every naked
  # `IntLiteral` value; the key is already a `String` literal.
  counts : Hash<String, UInt64> = { "bytes" => 3, "extra" => 1 }
  counts["extra"]
end

  include Loggable

  id : String

  def initialize( @id : String )
  end

# Everything an iterable gets for free once it can say how to visit its
# elements. `T` is the element type, carried through the `include` — an
# `Array<T>` includes `Enumerable<T>`, so `map` on it really does yield an
# `Array` of whatever the block returns.

mixin Enumerable<T>
  abstract def each( &block : T -> Void ) -> Void

  def map<U>( &block : T -> U ) -> Array<U>
    result = Array<U>.new
    each do | item |
      result.push( block.call( item ) )
    end
    result
  end

  def filter( &block : T -> Bool ) -> Array<T>
    result = Array<T>.new
    each do | item |
      if block.call( item )
        result.push( item )
      end
    end
    result
  end

  def to_array -> Array<T>
    result = Array<T>.new
    each do | item |
      result.push( item )
    end
    result
  end

  def count -> UInt64
    total = 0_u64
    each do | item |
      total += 1
    end
    total
  end

  def any?( &block : T -> Bool ) -> Bool
    found = false
    each do | item |
      if block.call( item )
        found = true
        break
      end
    end
    found
  end

  def all?( &block : T -> Bool ) -> Bool
    ok = true
    each do | item |
      if not block.call( item )
        ok = false
        break
      end
    end
    ok
  end

  def each_with_index( &block : ( T, UInt64 ) -> Void ) -> Void
    idx = 0_u64
    each do | item |
      block.call( item, idx )
      idx += 1
    end
  end
end

# A single decimal-rendering algorithm, written once against `self`.
#
# InterpLowering rewrites `"n = #{ x }"` into `"n = " + x.to_string`, so every
# type that can appear inside an interpolation must answer `to_string`. Rather
# than one hand-written body per integer width, the algorithm lives here and
# every width simply `include Stringable`.
#
# It is deliberately expressed in `self` arithmetic and String concatenation
# only — no raw byte store into a `Pointer<UInt8>` buffer. A byte store would
# force a `self` value into a `UInt8` slot, and Volt has no conversion between
# integer widths, so the store would be a genuine type error the moment
# assignability is checked. Concatenating one-character literals keeps every
# operand at exactly one type.

mixin Stringable
  # `self / 10` and `self % 10` are the whole algorithm; a type that includes
  # Stringable must therefore be arithmetic. Declared here as contracts rather
  # than assumed, so a non-arithmetic includer fails at the mixin, not deep
  # inside `to_string`.
  include Comparable

  abstract def /( other : self ) -> self
  abstract def %( other : self ) -> self
  abstract def -( other : self ) -> self

  # Ten explicit branches instead of an index into "0123456789": subscripting a
  # String takes a UInt64 and yields a UInt8, so it would mix widths with `self`
  # at two points. Ten comparisons cost nothing and stay width-honest.
  def digit_char -> String
    d = self % 10
    return "0" if d == 0
    return "1" if d == 1
    return "2" if d == 2
    return "3" if d == 3
    return "4" if d == 4
    return "5" if d == 5
    return "6" if d == 6
    return "7" if d == 7
    return "8" if d == 8
    "9"
  end

  def to_string -> String
    return "0" if self == 0

    # False for every unsigned width — the branch is dead there, not wrong.
    negative = self < 0
    n = negative ? 0 - self : self

    result = ""
    while n > 0
      result = n.digit_char + result
      n = n / 10
    end

    negative ? "-" + result : result
  end
end

@[Literal( ArrayLit )]
struct Array<T>
  include Enumerable<T>

  getter buffer : Pointer<T>
  getter size : UInt64
  getter capacity : UInt64

  def initialize( initial_capacity : UInt64 = 8 )
    @size = 0
    @capacity = initial_capacity
    @buffer = Pointer<T>.malloc( @capacity )
  end

  def empty? -> Bool
    @size == 0
  end

  def []( index : UInt64 ) -> T
    *( @buffer + index )
  end

  def []=( index : UInt64, value : T ) -> Void
    *( @buffer + index ) = value
  end

  def push( value : T ) -> Void
    if @size >= @capacity
      new_cap = @capacity == 0 ? 8 : @capacity * 2
      new_buf = Pointer<T>.malloc( new_cap )
      i = 0_u64
      while i < @size
        *( new_buf + i ) = *( @buffer + i )
        i += 1
      end
      @buffer.free
      @buffer = new_buf
      @capacity = new_cap
    end

    *( @buffer + @size ) = value
    @size += 1
  end

  def pop -> T
    if @size > 0
      @size -= 1
      *( @buffer + @size )
    end
  end

  def each( &block : T -> Void ) -> Void
    i = 0_u64
    while i < @size
      block.call( *( @buffer + i ) )
      i += 1
    end
  end

  def clear -> Void
    @size = 0
  end
end

@[Primitive( "u8", 8 ), Literal( CharLiteral )]
struct Char
  include Comparable
  include Hashable

  def <=>( other : Char ) -> Int32
    if self < other
      -1
    elsif self > other
      1
    else
      0
    end
  end

  def hash -> UInt64
    self
  end

  # One byte, not a decimal rendering: `"#{ 'A' }"` must read "A", not "65".
  # Char therefore cannot `include Stringable` — its digits are not its value.
  def to_string -> String
    buf = Pointer<UInt8>.malloc( 2_u64 )
    *( buf ) = self
    *( buf + 1_u64 ) = 0_u8
    String.new( buf, 1_u64 )
  end
end

@[Literal( StringLiteral )]
struct String
  include Comparable
  include Hashable

  getter data : Pointer<UInt8>
  getter size : UInt64

  def initialize( @data : Pointer<UInt8>, @size : UInt64 )
  end

  def empty? -> Bool
    @size == 0
  end

  def []( index : UInt64 ) -> UInt8
    *( @data + index )
  end

  def []=( index : UInt64, value : UInt8 ) -> Void
    *( @data + index ) = value
  end

  def <=>( other : String ) -> Int32
    min_size = @size < other.size ? @size : other.size
    cmp = memcmp( @data, other.data, min_size )
    return cmp unless cmp == 0

    if @size < other.size
      -1
    elsif @size > other.size
      1
    else
      0
    end
  end

  def +( other : String ) -> String
    total = @size + other.size
    buf = Pointer<UInt8>.malloc( total + 1_u64 )
    memcpy( buf, @data, @size )
    memcpy( buf + @size, other.data, other.size )
    *( buf + total ) = 0_u8
    String.new( buf, total )
  end

  def hash -> UInt64
    h = 5381
    i = 0_u64
    while i < @size
      h = ( ( h << 5 ) + h + *( @data + i ) ) & 72057594037927935
      i += 1
    end
    h
  end

  def to_string -> String
    self
  end

  def trim -> String
    start = 0_u64
    while start < @size
      b = *( @data + start )
      break unless b == 32_u8 or b == 9_u8 or b == 10_u8 or b == 13_u8
      start += 1
    end

    finish = @size
    while finish > start
      b = *( @data + ( finish - 1_u64 ) )
      break unless b == 32_u8 or b == 9_u8 or b == 10_u8 or b == 13_u8
      finish -= 1
    end

    len = finish - start
    buf = Pointer<UInt8>.malloc( len + 1_u64 )
    memcpy( buf, @data + start, len )
    *( buf + len ) = 0_u8
    String.new( buf, len )
  end

  def downcase -> String
    buf = Pointer<UInt8>.malloc( @size + 1_u64 )
    i = 0_u64
    while i < @size
      b = *( @data + i )
      if b >= 65_u8 and b <= 90_u8
        *( buf + i ) = b + 32_u8
      else
        *( buf + i ) = b
      end
      i += 1
    end
    *( buf + @size ) = 0_u8
    String.new( buf, @size )
  end

  def upcase -> String
    buf = Pointer<UInt8>.malloc( @size + 1_u64 )
    i = 0_u64
    while i < @size
      b = *( @data + i )
      if b >= 97_u8 and b <= 122_u8
        *( buf + i ) = b - 32_u8
      else
        *( buf + i ) = b
      end
      i += 1
    end
    *( buf + @size ) = 0_u8
    String.new( buf, @size )
  end

  def prefix( pre : String ) -> String
    pre + self
  end
end
